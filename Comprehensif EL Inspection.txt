# 1. DATA LOADING & VALIDATION
# ═════════════════════════════════════════════════════════════════════════════

def load_and_validate_data(file_path: str) -> pd.DataFrame:
    """
    Load EL inspection dataset and perform validation
    """
    print("="*80)
    print("LOADING AND VALIDATING DATASET")
    print("="*80)

    # Load CSV
    df = pd.read_csv(file_path)

    print(f"\n✓ Dataset loaded successfully")
    print(f"  Total records    : {len(df)}")
    print(f"  Total columns    : {len(df.columns)}")
    print(f"  Columns          : {list(df.columns)}")

    # Validate module count
    if len(df) != cfg.N_MODULES_TOTAL:
        print(f"\n⚠ Warning: Expected {cfg.N_MODULES_TOTAL} modules, found {len(df)}")
    else:
        print(f"\n✓ Module count matches configuration (204 modules)")

    # Check for missing values
    missing = df.isnull().sum()
    if missing.sum() > 0:
        print(f"\n⚠ Missing values detected:")
        print(missing[missing > 0])
    else:
        print(f"\n✓ No missing values")

    # Validate data types
    print(f"\n✓ Data types:")
    print(df.dtypes)

    # Convert inspection date to datetime
    df['Inspection_Date'] = pd.to_datetime(df['Inspection_Date'], format='%m/%d/%Y', errors='coerce')

    # Validate severity levels
    unique_severity = df['Severity_Level'].unique()
    print(f"\n✓ Unique severity levels: {list(unique_severity)}")

    # Validate power ratings
    if df['Rated_Power'].nunique() > 1:
        print(f"\n⚠ Multiple rated power values found: {df['Rated_Power'].unique()}")
    else:
        print(f"\n✓ Rated power consistent: {df['Rated_Power'].iloc[0]} Wp")

    return df

# 3. SEVERITY DISTRIBUTION ANALYSIS
# ═════════════════════════════════════════════════════════════════════════════

def analyze_severity_distribution(df: pd.DataFrame) -> pd.DataFrame:
    """
    Analyze distribution of modules by severity level
    """
    print("\n" + "="*80)
    print("SEVERITY LEVEL DISTRIBUTION")
    print("="*80)

    severity_dist = df['Severity_Level'].value_counts().reindex(cfg.SEVERITY_ORDER).fillna(0).astype(int)
    severity_pct = (severity_dist / len(df)) * 100

    severity_summary = pd.DataFrame({
        'Count': severity_dist,
        'Percentage': severity_pct,
        'Total_Rated_Power_W': [df[df['Severity_Level'] == level]['Rated_Power'].sum()
                                 for level in severity_dist.index],
        'Total_Actual_Power_W': [df[df['Severity_Level'] == level]['Actual_Power'].sum()
                                  for level in severity_dist.index],
        'Avg_Power_Loss_Pct': [df[df['Severity_Level'] == level]['Power_Loss'].mean()
                               for level in severity_dist.index],
        'Avg_Crack_Area_Pct': [df[df['Severity_Level'] == level]['Crack_Area'].mean()
                               for level in severity_dist.index],
    })

    print(f"\n{'Severity Level':<15} {'Count':>8} {'%':>8} {'Rated (W)':>12} {'Actual (W)':>12} {'Avg Loss %':>10}")
    print("-"*80)
    for idx, row in severity_summary.iterrows():
        print(f"{idx:<15} {int(row['Count']):>8} {row['Percentage']:>7.1f}% "
              f"{row['Total_Rated_Power_W']:>12,.0f} {row['Total_Actual_Power_W']:>12,.0f} "
              f"{row['Avg_Power_Loss_Pct']:>9.2f}")

    # Calculate quality metrics
    good_modules = severity_summary.loc[['Good', 'Minor'], 'Count'].sum()
    degraded_modules = severity_summary.loc[['Moderate', 'Severe', 'Critical', 'Catastrophic'], 'Count'].sum()

    print(f"\n✓ Quality Metrics:")
    print(f"  Good/Minor Modules      : {good_modules} ({(good_modules/len(df))*100:.1f}%)")
    print(f"  Degraded Modules        : {degraded_modules} ({(degraded_modules/len(df))*100:.1f}%)")

    return severity_summary

# 4. CRACK TYPE ANALYSIS
# ═════════════════════════════════════════════════════════════════════════════

def analyze_crack_types(df: pd.DataFrame) -> pd.DataFrame:
    """
    Analyze distribution and impact of different crack types
    """
    print("\n" + "="*80)
    print("CRACK TYPE ANALYSIS")
    print("="*80)

    crack_type_stats = df.groupby('Crack_Type').agg(
        Count=('ID_PV', 'count'),
        Avg_Power_Loss=('Power_Loss', 'mean'),
        Avg_Crack_Area=('Crack_Area', 'mean'),
        Avg_HotSpot_Area=('HotSpot_Area', 'mean'),
        Total_Actual_Power=('Actual_Power', 'sum'),
        Max_Severity=('Severity_Level', lambda x: max(x, key=lambda s: cfg.SEVERITY_ORDER.index(s) if s in cfg.SEVERITY_ORDER else -1))
    ).sort_values('Count', ascending=False)

    print(f"\nTop 10 Most Common Crack Types:")
    print(f"{'Crack Type':<50} {'Count':>8} {'Avg Loss %':>12} {'Avg Crack %':>12}")
    print("-"*85)
    for idx, row in crack_type_stats.head(10).iterrows():
        print(f"{idx:<50} {int(row['Count']):>8} {row['Avg_Power_Loss']:>11.2f}% {row['Avg_Crack_Area']:>11.2f}%")

    # Identify most critical crack types
    critical_types = df[df['Severity_Level'].isin(['Critical', 'Catastrophic'])]['Crack_Type'].value_counts()
    print(f"\n✓ Most Critical Crack Types (Critical/Catastrophic):")
    for crack_type, count in critical_types.head(5).items():
        print(f"  • {crack_type}: {count} modules")

    return crack_type_stats

# 5. STRING MAPPING & MISMATCH ANALYSIS
# ═════════════════════════════════════════════════════════════════════════════

def analyze_string_configuration(df: pd.DataFrame) -> pd.DataFrame:
    """
    Map modules to string configuration and calculate mismatch losses
    """
    print("\n" + "="*80)
    print("STRING CONFIGURATION & MISMATCH ANALYSIS")
    print("="*80)

    # Map modules to strings (sequential mapping)
    df_mapped = df.copy()
    df_mapped['String_ID'] = (df_mapped.index // cfg.N_MODULES_PER_STRING) + 1
    df_mapped['Position_in_String'] = (df_mapped.index % cfg.N_MODULES_PER_STRING) + 1

    # Calculate string-level parameters
    string_analysis = df_mapped.groupby('String_ID').agg(
        N_Modules=('ID_PV', 'count'),
        Total_Rated_Power=('Rated_Power', 'sum'),
        Total_Actual_Power_Sum=('Actual_Power', 'sum'),
        Min_Imp=('Imp_A', 'min'),
        Max_Imp=('Imp_A', 'max'),
        Avg_Imp=('Imp_A', 'mean'),
        Total_Vmp=('Vmp_V', 'sum'),
        Avg_Power_Loss=('Power_Loss', 'mean'),
        Worst_Severity=('Severity_Level', lambda x: max(x, key=lambda s: cfg.SEVERITY_ORDER.index(s) if s in cfg.SEVERITY_ORDER else -1)),
        Total_Cracks=('Num_Cracks', 'sum'),
        Max_Crack_Area=('Crack_Area', 'max'),
    ).reset_index()

    # Calculate actual string power (limited by lowest current module in series)
    # P_string = V_string_total × I_min
    string_analysis['String_Actual_Power_Calculated'] = (
        string_analysis['Total_Vmp'] * string_analysis['Min_Imp']
    )

    # Calculate mismatch loss
    string_analysis['Mismatch_Loss_W'] = (
        string_analysis['Total_Actual_Power_Sum'] - string_analysis['String_Actual_Power_Calculated']
    )
    string_analysis['Mismatch_Loss_Pct'] = (
        string_analysis['Mismatch_Loss_W'] / string_analysis['Total_Actual_Power_Sum'] * 100
    )

    print(f"\n{'String':>6} {'Modules':>8} {'Rated (W)':>12} {'Actual Sum (W)':>15} {'String Calc (W)':>16} {'Mismatch %':>10} {'Worst Sev':>12}")
    print("-"*95)
    for _, row in string_analysis.iterrows():
        print(f"{int(row['String_ID']):>6} {int(row['N_Modules']):>8} "
              f"{row['Total_Rated_Power']:>12,.0f} {row['Total_Actual_Power_Sum']:>15,.1f} "
              f"{row['String_Actual_Power_Calculated']:>16,.1f} {row['Mismatch_Loss_Pct']:>9.2f}% "
              f"{row['Worst_Severity']:>12}")

    # Summary statistics
    total_mismatch = string_analysis['Mismatch_Loss_W'].sum()
    avg_mismatch_pct = string_analysis['Mismatch_Loss_Pct'].mean()

    print(f"\n✓ String Mismatch Summary:")
    print(f"  Total Mismatch Loss       : {total_mismatch:,.1f} W ({total_mismatch/1000:.2f} kW)")
    print(f"  Average Mismatch per String: {avg_mismatch_pct:.2f}%")
    print(f"  Worst String (Mismatch)   : String #{string_analysis.loc[string_analysis['Mismatch_Loss_Pct'].idxmax(), 'String_ID']:.0f} "
          f"({string_analysis['Mismatch_Loss_Pct'].max():.2f}%)")

    return string_analysis, df_mapped

# 6. CORRELATION ANALYSIS
# ═════════════════════════════════════════════════════════════════════════════

def analyze_correlations(df: pd.DataFrame):
    """
    Analyze correlations between defect parameters and power loss
    """
    print("\n" + "="*80)
    print("CORRELATION ANALYSIS")
    print("="*80)

    # Select numerical columns for correlation
    corr_cols = ['Power_Loss', 'Crack_Area', 'HotSpot_Area', 'Num_Cracks',
                 'Actual_Power', 'Vmp_V', 'Imp_A', 'Voc_V', 'Isc_A']
    corr_matrix = df[corr_cols].corr()

    print(f"\nCorrelation with Power Loss:")
    print(f"{'Parameter':<20} {'Correlation':>12}")
    print("-"*35)
    for col in corr_cols:
        if col != 'Power_Loss':
            corr_val = corr_matrix.loc['Power_Loss', col]
            print(f"{col:<20} {corr_val:>12.4f}")

    # Identify strong correlations
    strong_corr = corr_matrix['Power_Loss'].abs().sort_values(ascending=False)
    print(f"\n✓ Strongest Correlations with Power Loss (|r| > 0.5):")
    for param, corr in strong_corr.items():
        if param != 'Power_Loss' and corr > 0.5:
            strength = "Very Strong" if corr > 0.7 else "Strong"
            print(f"  • {param}: r = {corr:.4f} ({strength})")

    return corr_matrix

    axes[1, 0].set_ylabel('Power Loss (%)')

    axes[1, 0].grid(True, alpha=0.3)
